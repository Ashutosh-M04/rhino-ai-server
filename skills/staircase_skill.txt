def make_dotnet_request(url, payload_dict, timeout_ms=60000):
System.Net.ServicePointManager.SecurityProtocol = System.Net.SecurityProtocolType.Tls12
request = System.Net.WebRequest.Create(url)
request.Method = "POST"
request.Timeout = timeout_ms
request.ContentType = "application/json"
request.UserAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"

if payload_dict:
    json_bytes = System.Text.Encoding.UTF8.GetBytes(json.dumps(payload_dict))
    request.ContentLength = json_bytes.Length
    req_stream = request.GetRequestStream()
    req_stream.Write(json_bytes, 0, json_bytes.Length)
    req_stream.Close()
    
response = request.GetResponse()
res_stream = response.GetResponseStream()
reader = System.IO.StreamReader(res_stream)
res_text = reader.ReadToEnd()
reader.Close()
response.Close()

return json.loads(res_text)

# --- LICENSE MANAGEMENT ---
def load_license_data(self):
    if os.path.exists(LICENSE_FILE):
        try:
            with open(LICENSE_FILE, "r") as f:
                data = json.load(f)
                self.current_session_key = data.get("key", "")
                self.session_count = data.get("sessions", 0)
                
                if self.current_session_key and self.session_count < 1000:
                    self.is_licensed = True
                    self.session_count += 1
                    self.save_license_data()
                    self.lock_license_tab()
                    self.run_button.Enabled = True
                    self.report_button.Enabled = True
                    self.no_output_button.Enabled = True
                else:
                    self.is_licensed = False
                    self.session_count = 0
                    self.run_button.Enabled = False
                    self.report_button.Enabled = False
                    self.no_output_button.Enabled = False
        except:
            pass

def save_license_data(self):
    with open(LICENSE_FILE, "w") as f:
        json.dump({"key": self.current_session_key, "sessions": self.session_count}, f)

def lock_license_tab(self):
    self.license_box.Enabled = False
    self.validate_button.Enabled = False
    sessions_left = 1001 - self.session_count
    self.license_box.PlaceholderText = "Active! ({} sessions remaining)".format(sessions_left)
    self.license_box.Text = ""

def on_validate_click(self, sender, e):
    key_input = self.license_box.Text
    if not key_input: return
    
    try:
        payload = {
            "key": key_input,
            "machine_id": get_machine_id()
        }
        
        result = make_dotnet_request(LICENSE_URL, payload)
        
        if result.get("status") == "valid":
            self.current_session_key = key_input
            self.session_count = 1
            self.is_licensed = True
            self.save_license_data()
            self.lock_license_tab()
            self.run_button.Enabled = True
            self.report_button.Enabled = True
            self.no_output_button.Enabled = True
            Rhino.UI.Dialogs.ShowMessage("License Validated! Locked to this machine.", "Success")
        else:
            Rhino.UI.Dialogs.ShowMessage(result.get("message", "Invalid or Expired Key."), "Error")
            
    except Exception as err:
        Rhino.UI.Dialogs.ShowMessage("Could not connect to License Server.\n{}".format(err), "Network Error")

def process_request(self, prompt, error_feedback, w_val, r_val, h_val, img_paths, attempt):
    if attempt > 5:
        self.update_status("Max attempts reached. AI could not fix the error.")
        self.reenable_run()
        return
        
    try:
        if error_feedback:
            if "USER REPORTED" in error_feedback:
                self.update_status("Sending User Report to Antigravity IDE...")
            else:
                self.update_status("Attempt {}/5: Sending Python error back to AI...".format(attempt))
        else:
            self.update_status("analyzing...")
            
        time.sleep(0.5) 
        self.update_status("sending the request...")
        
        safe_error_flag = ERROR_FLAG.replace("\\", "\\\\")
        safe_done_flag = DONE_FLAG.replace("\\", "\\\\")
        
        error_injection = """
        
        CRITICAL SYSTEM INSTRUCTION:
        You MUST structure your Python script exactly like the boilerplate below. 
        Put all your logic inside the 'build_geometry()' function. DO NOT deviate from this layout, or Rhino will crash.
        
        import traceback
        import rhinoscriptsyntax as rs
        import scriptcontext as sc
        import Rhino

        def build_geometry():
            # --- YOUR GENERATED CODE GOES HERE ---
            pass

        try:
            build_geometry()
            with open(r'{done}', 'w') as f: f.write('success')
        except Exception as e:
            with open(r'{error}', 'w') as f: f.write(traceback.format_exc())
        """.format(done=safe_done_flag, error=safe_error_flag)
        
        enhanced_prompt = prompt + error_injection
        
        payload = {
            "session_id": "Session_UI",
            "user_prompt": enhanced_prompt,
            "previous_error": error_feedback if error_feedback else "",
            "dimensions": {
                "tread_width": w_val,
                "riser_height": r_val,
                "total_height": h_val
            },
            "images": img_paths
        }
        
        res_server_body = make_dotnet_request(GENERATE_URL, payload, timeout_ms=120000)
        instructions = res_server_body.get("antigravity_instructions")
        
        self.update_status("Executing the request (2/3) - Updating Payload...")
        
        if os.path.exists(PAYLOAD_FILE):
            try:
                os.remove(PAYLOAD_FILE)
                time.sleep(0.2) 
            except Exception as e:
                pass
        
        with open(PAYLOAD_FILE, "w") as f:
            json.dump({
                "session_id": "Session_UI", 
                "prompt_instructions": instructions,
                "images": img_paths
            }, f, indent=2)
        
        self.update_status("Executing the request (3/3)")
        time.sleep(0.5)
        
        self.update_status("waiting for Antigravity IDE...")
        self.watch_for_completion(prompt, w_val, r_val, h_val, img_paths, attempt)

    except Exception as e:
        self.update_status("Error in sending stage: {}".format(e))
        self.reenable_run()

def watch_for_completion(self, prompt, w_val, r_val, h_val, img_paths, attempt):
    timeout = 600 
    start_time = time.time()
    
    while time.time() - start_time < timeout:
        if os.path.exists(ERROR_FLAG):
            time.sleep(0.5) 
            with open(ERROR_FLAG, "r") as f:
                err_msg = f.read()
            
            self.update_status("Python error caught! Fixing (Attempt {}/5)...".format(attempt))
            os.remove(ERROR_FLAG)
            if os.path.exists(DONE_FLAG):
                os.remove(DONE_FLAG) 
            
            time.sleep(1)
            self.process_request(prompt, err_msg, w_val, r_val, h_val, img_paths, attempt + 1)
            return
            
        if os.path.exists(DONE_FLAG):
            self.update_status("The script has been successfully executed.")
            os.remove(DONE_FLAG)
            self.reenable_run()
            return
            
        time.sleep(1)
        
    self.update_status("Error: Antigravity IDE timeout after 10 minutes.")
    self.reenable_run()
